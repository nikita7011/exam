Redux Toolkit Configuration

slice.ts (State slice setup) 
import { createSlice, PayloadAction } from "@reduxjs/toolkit";

interface Item {
  id: number;
  name: string;
  price: number;
  quantity?: number;
}

interface State {
  items: Item[];
  total: number;
}

const initialState: State = {
  items: [],
  total: 0
};

const slice = createSlice({
  name: "app",
  initialState,
  reducers: {
    add: (state, action: PayloadAction) => {
      state.items.push(action.payload);
      state.total += action.payload.price;
    },

    remove: (state, action: PayloadAction) => {
      const item = state.items.find(i => i.id === action.payload);
      if (item) state.total -= item.price;

      state.items = state.items.filter(i => i.id !== action.payload);
    },

    clear: (state) => {
      state.items = [];
      state.total = 0;
    }
  }
});

export const { add, remove, clear } = slice.actions;
export default slice.reducer;

Zod Schema Validation

Zod Definition schema.ts 
import { z } from "zod";

const schema = z.object({
  name: z.string().min(2),
  email: z.string().email(),
  password: z.string().min(6),
  age: z.number().min(18),
  phone: z.string().regex(/^[0-9]{10}$/),
  address: z.object({
    city: z.string(),
    state: z.string(),
  }),
  categories: z.array(z.string()).min(1),
  terms: z.literal(true),
}).refine((data) => data.password === data.confirmPassword, {
  path: ["confirmPassword"],
  message: "Passwords must match",
});

type FormData = z.infer<typeof schema>;

Yup Schema Validation

Yup Definition schema.ts 
import * as yup from "yup";

type FormData = {
  field1: string;
  field2: string;
  field3?: string;
  option: string;
  terms: boolean;
};

const schema = yup.object({
  field1: yup.string().required(),

  field2: yup
    .string()
    .matches(/regex/)
    .required(),

  field3: yup.string().when("option", {
    is: "value",
    then: (s) => s.required(),
  }),

  option: yup
    .string()
    .oneOf(["value1", "value2"])
    .required(),

  terms: yup
    .boolean()
    .oneOf([true]),
});

const typeDefs = `#graphql
  type Child {
    id: ID
    name: String
  }

  type Parent {
    id: ID
    name: String
    children: [Child]
  }

  type Query {
    parents: [Parent]
    parent(id: ID!): Parent
  }

  type Mutation {
    addParent(name: String!): Parent
    addChild(parentId: ID!, name: String!): Child
  }
`;

const resolvers = {
  Query: {
    parents: () => parentsData,

    parent: (_, { id }) =>
      parentsData.find(p => p.id == id),
  },

  Mutation: {
    addParent: (_, { name }) => {
      return { id: Date.now(), name, children: [] };
    },

    addChild: (_, { parentId, name }) => {
      const parent = parentsData.find(p => p.id == parentId);

      const newChild = { id: Date.now(), name };

      if (parent) {
        parent.children.push(newChild);
      }

      return newChild;
    },
  },

  Parent: {
    children: (parent) => parent.children,
  },

import { useForm } from "react-hook-form";

type FormData = {
  email: string;
  password: string;
};

export default function Login() {
  const {
    register,
    handleSubmit,
    setError,
    formState: { errors },
  } = useForm<FormData>();

  const onSubmit = async (data: FormData) => {
    await new Promise((r) => setTimeout(r, 1000));

    if (data.email !== "admin@gmail.com") {
      setError("email", { message: "Invalid email or password" });
    } else {
      console.log("Login Success");
    }
  };

  return (
    <form onSubmit={handleSubmit(onSubmit)}>
      <input {...register("email")} />
      {errors.email && <p>{errors.email.message}</p>}

      <input type="password" {...register("password")} />

      <button type="submit">Login</button>
    </form>
  );
}
